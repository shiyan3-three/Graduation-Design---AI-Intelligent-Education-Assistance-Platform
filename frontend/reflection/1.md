import React, { useState, useRef, useEffect } from 'react';
import { Layout, Button, Input, Spin, Avatar, Typography, Tooltip } from 'antd';
import { 
  MenuFoldOutlined, 
  MenuUnfoldOutlined, 
  SendOutlined, 
  PlusOutlined, 
  UserOutlined, 
  RobotOutlined,
  HistoryOutlined
} from '@ant-design/icons';

const { Header, Sider, Content } = Layout;
const { Text } = Typography;

// 严格对齐后台设计的消息数据结构
interface Message {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  tools_used?: string[];
}

const ChatPage: React.FC = () => {
  // UI 状态
  const [collapsed, setCollapsed] = useState(false);
  const [inputVal, setInputVal] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  // 数据状态
  const [messages, setMessages] = useState<Message[]>([
    { id: '1', role: 'assistant', content: '同学你好！我是你的智能教育辅导 Agent。今天想复习哪个知识点？' }
  ]);

  // 引用，用于自动滚动到底部
  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    // 每次消息更新，平滑滚动到最新消息
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  // 发送消息核心逻辑
  const handleSend = () => {
    // 防抖/防连发：内容为空或正在加载时直接拦截
    if (!inputVal.trim() || isLoading) return;

    const newUserMsg: Message = { 
      id: Date.now().toString(), 
      role: 'user', 
      content: inputVal.trim() 
    };
    
    setMessages(prev => [...prev, newUserMsg]);
    setInputVal(''); // 清空输入框
    setIsLoading(true); // 开启防抖和 Loading 状态
    
    // Mock 后端非流式 API 调用
    setTimeout(() => {
      const newAiMsg: Message = { 
        id: (Date.now() + 1).toString(), 
        role: 'assistant', 
        content: '这是一个模拟的非流式回答。根据你的问题，我已经调用了本地结构化题库为你找到了相关解析。',
        tools_used: ['question_tool'] // 模拟后台触发了 Fallback 策略中的工具
      };
      setMessages(prev => [...prev, newAiMsg]);
      setIsLoading(false); // 解除锁定
    }, 2000);
  };

  // 快捷键处理
  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    // 仅按 Enter 且没有按组合键时 -> 发送
    if (e.key === 'Enter' && !e.ctrlKey && !e.shiftKey) {
      e.preventDefault(); // 阻止默认的换行行为
      handleSend();
    }
    // Ctrl + Enter 或 Shift + Enter -> 允许系统默认的换行行为，无需额外干预
  };

  return (
    <Layout className="h-screen bg-gray-50 overflow-hidden">
      {/* --- 左侧边栏 --- */}
      <Sider 
        trigger={null} 
        collapsible 
        collapsed={collapsed}
        theme="light"
        className="shadow-sm border-r border-gray-200"
        width={260}
      >
        <div className="flex flex-col h-full">
          {/* Logo 与折叠按钮区 */}
          <div className="h-16 flex items-center justify-between px-4 border-b border-gray-100">
            {!collapsed && <span className="font-bold text-lg text-blue-600 truncate">AI 智能辅导平台</span>}
            <Button 
              type="text" 
              icon={collapsed ? <MenuUnfoldOutlined /> : <MenuFoldOutlined />} 
              onClick={() => setCollapsed(!collapsed)} 
              className="text-gray-500 hover:text-blue-600"
            />
          </div>

          {/* 新建会话按钮 */}
          <div className="p-4">
            <Button 
              type="primary" 
              block 
              icon={<PlusOutlined />} 
              className={`flex items-center justify-center ${collapsed ? 'px-0' : ''}`}
              title={collapsed ? '新建会话' : ''}
            >
              {!collapsed && '新建辅导会话'}
            </Button>
          </div>
    
          {/* 历史记录 Mock 占位 */}
          <div className="flex-1 overflow-y-auto px-2">
            {!collapsed && <div className="text-xs text-gray-400 px-2 mb-2">最近的会话</div>}
            <div className="flex items-center px-3 py-2 cursor-pointer hover:bg-gray-100 rounded-lg text-gray-600 mb-1">
              <HistoryOutlined className="mr-2" />
              {!collapsed && <span className="truncate text-sm">微积分洛必达法则</span>}
            </div>
            <div className="flex items-center px-3 py-2 cursor-pointer hover:bg-gray-100 rounded-lg text-gray-600">
              <HistoryOutlined className="mr-2" />
              {!collapsed && <span className="truncate text-sm">Python 字典推导式</span>}
            </div>
          </div>
        </div>
      </Sider>
    
      {/* --- 右侧主界面 --- */}
      <Layout>
        {/* 顶部 Header 可留空或放用户信息 */}
        <Header className="bg-white border-b border-gray-200 h-16 flex items-center px-6" />
    
        {/* 核心对话区 */}
        <Content className="flex flex-col bg-white">
          {/* 消息列表区 */}
          <div className="flex-1 overflow-y-auto p-6 space-y-6">
            {messages.map((msg) => (
              <div 
                key={msg.id} 
                className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
              >
                <div className={`flex max-w-[80%] ${msg.role === 'user' ? 'flex-row-reverse' : 'flex-row'}`}>
                  {/* 头像 */}
                  <Avatar 
                    icon={msg.role === 'user' ? <UserOutlined /> : <RobotOutlined />} 
                    className={`shrink-0 ${msg.role === 'user' ? 'ml-4 bg-blue-500' : 'mr-4 bg-purple-500'}`}
                  />
                  
                  {/* 消息气泡 */}
                  <div className={`flex flex-col ${msg.role === 'user' ? 'items-end' : 'items-start'}`}>
                    <div 
                      className={`px-4 py-3 rounded-2xl shadow-sm text-[15px] leading-relaxed
                        ${msg.role === 'user' 
                          ? 'bg-blue-600 text-white rounded-tr-sm' 
                          : 'bg-gray-100 text-gray-800 rounded-tl-sm'
                        }
                      `}
                    >
                      {msg.content}
                    </div>
    
                    {/* 工具调用状态展示 (针对 Agent) */}
                    {msg.tools_used && msg.tools_used.length > 0 && (
                      <div className="text-xs text-gray-400 mt-1 flex items-center">
                        <RobotOutlined className="mr-1" /> 已调用 MCP 工具: {msg.tools_used.join(', ')}
                      </div>
                    )}
                  </div>
                </div>
              </div>
            ))}
    
            {/* 加载/思考状态占位 */}
            {isLoading && (
              <div className="flex justify-start">
                <div className="flex max-w-[80%] flex-row items-center">
                  <Avatar icon={<RobotOutlined />} className="shrink-0 mr-4 bg-purple-500 opacity-50" />
                  <div className="px-4 py-3 bg-gray-50 text-gray-500 rounded-2xl rounded-tl-sm shadow-sm flex items-center">
                    <Spin size="small" className="mr-2" />
                    <Text type="secondary" className="text-sm">Agent 正在思考并检索知识...</Text>
                  </div>
                </div>
              </div>
            )}
            
            {/* 自动滚动锚点 */}
            <div ref={messagesEndRef} />
          </div>
    
          {/* 底部输入区 */}
          <div className="p-4 bg-white border-t border-gray-100">
            <div className="max-w-4xl mx-auto relative">
              <Input.TextArea
                value={inputVal}
                onChange={(e) => setInputVal(e.target.value)}
                onKeyDown={handleKeyDown}
                placeholder="在此输入您的问题 (Enter 发送，Ctrl+Enter 换行)"
                autoSize={{ minRows: 2, maxRows: 6 }}
                // 圆角定制与移除 antd 默认聚焦蓝框
                className="rounded-2xl py-3 pr-14 text-[15px] shadow-sm resize-none focus:ring-2 focus:ring-blue-100"
              />
              <Tooltip title="发送 (Enter)">
                <Button
                  type="primary"
                  shape="circle"
                  icon={<SendOutlined />}
                  onClick={handleSend}
                  loading={isLoading}
                  disabled={!inputVal.trim()}
                  // 绝对定位在输入框右下角
                  className="absolute right-3 bottom-3"
                />
              </Tooltip>
            </div>
            <div className="text-center mt-2">
              <Text type="secondary" className="text-xs">
                AI 回复由大模型 API 与统一 MCP Server 提供支持
              </Text>
            </div>
          </div>
        </Content>
      </Layout>
    </Layout>
      );
    };
    
    export default ChatPage;
