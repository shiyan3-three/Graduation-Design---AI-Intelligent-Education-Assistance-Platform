    <script>
        function switchTo(target) {
            const login = document.getElementById('login');
            const register = document.getElementById('register');
            
            if (target === 'register') {
                login.classList.replace('active', 'exit');
                // 延迟一下再让另一个入场，防止重叠碰撞
                setTimeout(() => { register.classList.replace('exit', 'active'); }, 100);
            } else {
                register.classList.replace('active', 'exit');
                setTimeout(() => { login.classList.replace('exit', 'active'); }, 100);
            }
        }
    </script>
