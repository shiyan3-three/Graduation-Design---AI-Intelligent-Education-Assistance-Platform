const mockQuestions = [
    {
        id: 1,

        ],
        correctAnswer: "B",
        analysis: "正确答案是 <strong>B</strong>。<br>在这里，“小明”是借书动作的受益者。“老师”是借书动作的执行者。使用「～てもらえませんか」是请求他人为自己做某事的最标准敬语表达。选项 A 的视角反了。选项 C 的意思是请求别人去借，逻辑不通。",
        status: "pending" // pending, completed
    },
    {
        id: 2,
        tag: "助词使用",
        title: "动作发生场所「で」与存在场所「に」辨析",
        type: "单选题",
        question: "请选择填入空白处最合适的助词：<br>「休みの日は、いつも公園（　　）本を読みます。」",
        options: [
            { id: "A", text: "に" },
            { id: "B", text: "で" },
            { id: "C", text: "へ" }
        ],
        correctAnswer: "B",
        analysis: "正确答案是 <strong>B</strong>。<br>「読む（阅读）」是一个具体的动作。表示动作发生的场所时，必须使用助词「で」。如果后接的是表示存在的动词（如 あります、います），才使用「に」。",
        status: "pending"
    },
    {
        id: 3,
        tag: "动词活用",
        title: "一类动词「て形」音便规则测验",
        type: "单选题",
        question: "动词「急ぐ（いそぐ）」的「て形」是哪一个？",
        options: [
            { id: "A", text: "いそって" },
            { id: "B", text: "いそいで" },
            { id: "C", text: "いそんで" }
        ],
        correctAnswer: "B",
        analysis: "正确答案是 <strong>B</strong>。<br>「急ぐ」是以「ぐ」结尾的一类动词，它遵循「イ音便」规则，即变为「い」加上浊音的「で」，所以变成「急いで」。",
        status: "pending"
    }
];

let currentIndex = 0; // 当前选中的题目索引

// ============================================================================
// 视图渲染逻辑
// ============================================================================

// 1. 渲染上下文 Banner
function renderContext() {
    const tags = [...new Set(mockQuestions.map(q => q.tag))];
    const container = document.getElementById('context-tags-container');
    container.innerHTML = tags.map(tag => `<span class="context-tag">${tag}</span>`).join('');
    document.getElementById('task-count').innerText = mockQuestions.length;
}

// 2. 渲染左侧任务列表
function renderTaskList() {
    const listHtml = mockQuestions.map((q, index) => {
        const isActive = index === currentIndex ? 'active' : '';
        const isCompleted = q.status === 'completed' ? 'completed' : '';
        const statusIcon = q.status === 'completed' ? '✔' : '●';
        
        return `
            <div class="task-item ${isActive} ${isCompleted}" onclick="selectQuestion(${index})">
                <div class="task-item-tag">${statusIcon} ${q.tag}</div>
                <div class="task-item-title">${q.title}</div>
            </div>
        `;
    }).join('');
    document.getElementById('task-list').innerHTML = listHtml;
}

// 3. 渲染右侧题目详情
function renderQuestion() {
    const q = mockQuestions[currentIndex];
    
    document.getElementById('q-type').innerText = q.type;
    document.getElementById('q-title').innerHTML = q.question;
    
    // 渲染选项
    const optionsHtml = q.options.map(opt => `
        <button class="opt-btn" onclick="handleAnswer('${opt.id}')" id="opt-${opt.id}">
            <span class="opt-letter">${opt.id}.</span>
            <span class="opt-text">${opt.text}</span>
        </button>
    `).join('');
    document.getElementById('options-list').innerHTML = optionsHtml;

    // 重置解析与按钮状态
    const analysisArea = document.getElementById('ai-analysis');
    analysisArea.style.display = 'none';
    
    const nextBtn = document.getElementById('btn-next');
    if(q.status === 'completed') {
        // 如果这道题已经做过了，直接展示解析
        showAnalysisAndHighlight(q);
        nextBtn.disabled = false;
        if(currentIndex === mockQuestions.length - 1) nextBtn.innerText = "完成练习";
    } else {
        nextBtn.disabled = true;
        nextBtn.innerText = "下一题";
    }
}

// ============================================================================
// 交互事件处理
// ============================================================================

// 切换左侧题目
function selectQuestion(index) {
    currentIndex = index;
    renderTaskList();
    renderQuestion();
}

// 处理点击选项（答题核心逻辑）
function handleAnswer(selectedOptId) {
    const q = mockQuestions[currentIndex];
    // 如果已经做过了，不重复触发
    if(q.status === 'completed') return;

    // 禁用所有选项
    const allOptBtns = document.querySelectorAll('.opt-btn');
    allOptBtns.forEach(btn => btn.disabled = true);

    // 高亮判断
    if(selectedOptId === q.correctAnswer) {
        document.getElementById(`opt-${selectedOptId}`).classList.add('correct');
    } else {
        document.getElementById(`opt-${selectedOptId}`).classList.add('wrong');
        // 同时标出正确答案
        document.getElementById(`opt-${q.correctAnswer}`).classList.add('correct');
    }

    // 标记为完成，展示解析，放开下一步按钮
    q.status = 'completed';
    showAnalysisAndHighlight(q);
    renderTaskList(); // 更新左侧列表的 ✔ 状态

    const nextBtn = document.getElementById('btn-next');
    nextBtn.disabled = false;
    if(currentIndex === mockQuestions.length - 1) {
        nextBtn.innerText = "完成练习";
    }
}

// 显示解析与回显高亮状态 (供已完成的题目重复查看)
function showAnalysisAndHighlight(q) {
    const allOptBtns = document.querySelectorAll('.opt-btn');
    allOptBtns.forEach(btn => btn.disabled = true);
    
    // 兜底高亮正确答案
    const correctBtn = document.getElementById(`opt-${q.correctAnswer}`);
    if(correctBtn) correctBtn.classList.add('correct');

    document.getElementById('ai-body').innerHTML = q.analysis;
    document.getElementById('ai-analysis').style.display = 'block';
}

// 下一题
function nextQuestion() {
    if(currentIndex < mockQuestions.length - 1) {
        currentIndex++;
        renderTaskList();
        renderQuestion();
    } else {
        alert("恭喜！本次靶向演练已全部完成。");
    }
}

// 跳过
function skipQuestion() {
    const q = mockQuestions[currentIndex];
    if(q.status !== 'completed') {
        q.status = 'completed'; // 跳过也算作处理完毕
    }
    nextQuestion();
}

    renderContext();
    renderTaskList();
    renderQuestion();
