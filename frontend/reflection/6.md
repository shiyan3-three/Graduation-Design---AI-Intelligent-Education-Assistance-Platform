        const mockTagsData = {
            "success": true, "message": "ok",
            "data": {
                "items": [
                    { "tag": "动词て形变形", "count": 12 },
                    { "tag": "助词に与で的区别", "count": 8 },

        // 模拟 API 请求 (延迟 1.5 秒展示骨架屏效果)
        function fetchAnalyticsData() {
            setTimeout(() => {
                renderHotspots(mockReportData.data.focus_areas);
                renderTags(mockTagsData.data.items);
            }, 1500);
        }
    
        // 渲染区块一
        function renderHotspots(areas) {
            document.getElementById('hotspot-skeleton').style.display = 'none';
            const container = document.getElementById('hotspot-data');
            
            if (!areas || areas.length === 0) {
                container.innerHTML = `<div class="empty-state">暂无热点数据</div>`;
                container.style.display = 'block';
                return;
            }
    
            let html = '';
            areas.forEach(item => {
                html += `
                    <div class="hotspot-card rank-${item.rank}">
                        <div class="rank-badge">No.${item.rank}</div>
                        <div class="hotspot-tag">${item.tag}</div>
                        <div class="hotspot-count">讨论 ${item.count} 次</div>
                    </div>
                `;
            });
            container.innerHTML = html;
            container.style.display = 'grid';
        }
    
        // 渲染区块二
        function renderTags(items) {
            document.getElementById('tags-skeleton').style.display = 'none';
            const container = document.getElementById('tags-data');
            const totalText = document.getElementById('tag-total-text');
    
            if (!items || items.length === 0) {
                container.innerHTML = `
                    <div class="empty-state">
                        <p>您还没有积累知识点标签</p>
                        <button class="primaryBtn" onclick="alert('跳转至 /chat')">去和 AI 对话</button>
                    </div>
                `;
                container.style.display = 'block';
                return;
            }
    
            totalText.innerText = `共 ${items.length} 个标签`;
            totalText.style.display = 'block';
    
            let html = '';
            items.forEach(item => {
                html += `
                    <div class="listItem">
                        <span class="itemTitle">${item.tag}</span>
                        <span class="badge primary">${item.count} 次</span>
                    </div>
                `;
            });
            container.innerHTML = html;
            container.style.display = 'flex';
        }
