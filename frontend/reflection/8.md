// ─── 模拟数据 ─────────────────────────────────────────────────────────────────
const ME = 9999;
const users = [
  { id:1, username:'张三', grade:'高一', subject:'数学', created_at:'2026-04-01', is_admin:false },
  { id:2, username:'李四', grade:'高二', subject:'物理', created_at:'2026-04-03', is_admin:false },
  { id:3, username:'王五', grade:'高三', subject:'英语', created_at:'2026-04-10', is_admin:false },
  { id:ME, username:'admin', grade:'管理', subject:'管理', created_at:'2026-04-22', is_admin:true },
];
const questions = [
  { id:1, subject:'数学', topic:'一次函数', question:'已知一次函数过点(1,3)和(2,5)，求该函数的解析式。', options:'A. y=2x+1\nB. y=x+2\nC. y=3x-1\nD. y=2x-1', answer:'A', editing:false, confirming:false },
  { id:2, subject:'物理', topic:'受力分析', question:'如图所示，物体在水平面上做匀速直线运动，以下说法正确的是？', options:'A. 合力为零\nB. 合力向右\nC. 没有摩擦力\nD. 重力大于支持力', answer:'A', editing:false, confirming:false },
  { id:3, subject:'数学', topic:'二次函数', question:'抛物线 y=x²-4x+3 的顶点坐标是？', options:'A. (2,-1)\nB. (2,1)\nC. (-2,-1)\nD. (4,3)', answer:'A', editing:false, confirming:false },
  { id:4, subject:'英语', topic:'动词变形', question:'请选出与"She __ (go) to school every day."最合适的词形。', options:'A. goes\nB. go\nC. going\nD. gone', answer:'A', editing:false, confirming:false },
];
let userConfirmId = null;
let showingAddForm = false;

// ─── Tab 切换 ─────────────────────────────────────────────────────────────────
function switchTab(name, btn) {
  document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('tab-' + name).classList.add('active');
  btn.classList.add('active');
}

// ─── 用户管理 ─────────────────────────────────────────────────────────────────
function renderUsers() {
  const q = (document.querySelector('#tab-users .inp-search') || {}).value || '';
  const tbody = document.getElementById('user-tbody');
  tbody.innerHTML = '';
  users.filter(u => !q || u.username.includes(q)).forEach((u, i) => {
    const isSelf = u.id === ME;
    const roleBadge = u.is_admin
      ? '<span class="badge badge-admin">管理员</span>'
      : '<span class="badge badge-user">普通用户</span>';
    const toggleLabel = u.is_admin ? '取消管理员' : '设为管理员';
    // main row
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${i+1}</td>
      <td><strong>${u.username}</strong></td>
      <td>${u.grade}</td><td>${u.subject}</td><td>${u.created_at}</td>
      <td>${roleBadge}</td>
      <td><div class="actions">
        <button class="btn-sm btn-blue" onclick="toggleAdmin(${u.id})">${toggleLabel}</button>
        <button class="btn-sm btn-danger${isSelf?' btn-disabled':''}" onclick="confirmDelUser(${u.id})"
          ${isSelf?'disabled':''}>删除</button>
      </div></td>`;
    tbody.appendChild(tr);
    // inline confirm
    if (userConfirmId === u.id) {
      const cr = document.createElement('tr');
      cr.className = 'row-confirm no-hover';
      cr.innerHTML = `<td colspan="7"><div class="confirm-area">
        <span class="confirm-text">⚠️ 确认删除用户「${u.username}」？此操作不可撤销，其全部会话和消息将一并删除。</span>
        <button class="btn-sm btn-ghost" onclick="cancelDelUser()">取消</button>
        <button class="btn-sm btn-danger" onclick="doDelUser(${u.id})">确认删除</button>
      </div></td>`;
      tbody.appendChild(cr);
    }
  });
}
function toggleAdmin(id) {
  const u = users.find(x => x.id===id); if(u) u.is_admin=!u.is_admin; renderUsers();
}
function confirmDelUser(id) { userConfirmId=id; renderUsers(); }
function cancelDelUser() { userConfirmId=null; renderUsers(); }
function doDelUser(id) {
  const i=users.findIndex(x=>x.id===id); if(i>-1) users.splice(i,1);
  userConfirmId=null; renderUsers();
}

// ─── 题库管理 ─────────────────────────────────────────────────────────────────
function renderQuestions() {
  const sf = document.getElementById('subjectF').value;
  const tf = document.getElementById('topicF').value.toLowerCase();
  const list = questions.filter(q => (!sf||q.subject===sf) && (!tf||q.topic.toLowerCase().includes(tf)));
  const tbody = document.getElementById('question-tbody');
  tbody.innerHTML = '';

  // 新增表单行
  if (showingAddForm) {
    const ar = document.createElement('tr');
    ar.className = 'row-add-outer no-hover';
    ar.innerHTML = `<td colspan="6" style="padding:14px 16px">
      <div class="edit-grid-add">
        <input class="inp" style="width:100%" placeholder="学科" id="na-subject">
        <input class="inp" style="width:100%" placeholder="专题" id="na-topic">
        <textarea class="ta" rows="3" placeholder="题目内容..." id="na-question"></textarea>
        <div class="ans-wrap"><span class="ans-label">答案</span>
          <input class="inp inp-ans" placeholder="A-D" id="na-answer"></div>
      </div>
      <textarea class="ta ta-opts" rows="4" placeholder="选项（合并格式）&#10;A. 选项内容&#10;B. 选项内容&#10;C. 选项内容&#10;D. 选项内容" id="na-options"></textarea>
      <div class="edit-actions">
        <button class="btn-sm btn-success" onclick="doAddQuestion()">✓ 保存</button>
        <button class="btn-sm btn-ghost" onclick="cancelAdd()">✕ 取消</button>
      </div>
    </td>`;
    tbody.appendChild(ar);
  }

  list.forEach((q, i) => {
    if (q.editing) {
      // 编辑行
      const er = document.createElement('tr');
      er.className = 'row-edit no-hover';
      er.innerHTML = `<td style="padding:14px 16px;vertical-align:top">${i+1}</td>
        <td colspan="5" style="padding:14px 16px">
          <div class="edit-grid">
            <textarea class="ta" rows="3" id="eq-${q.id}">${q.question}</textarea>
            <div class="ans-wrap"><span class="ans-label">答案</span>
              <input class="inp inp-ans" id="ea-${q.id}" value="${q.answer}" placeholder="A-D"></div>
          </div>
          <textarea class="ta ta-opts" rows="4" id="eo-${q.id}">${q.options}</textarea>
          <div class="edit-actions">
            <button class="btn-sm btn-success" onclick="doEditQ(${q.id})">✓ 保存修改</button>
            <button class="btn-sm btn-ghost" onclick="cancelEditQ(${q.id})">✕ 取消</button>
          </div>
        </td>`;
      tbody.appendChild(er);
    } else {
      // 普通行
      const tr = document.createElement('tr');
      tr.innerHTML = `<td>${i+1}</td>
        <td>${q.subject}</td><td>${q.topic}</td>
        <td><span class="trunc">${q.question}</span></td>
        <td><strong>${q.answer}</strong></td>
        <td><div class="actions">
          <button class="btn-sm btn-blue" onclick="startEditQ(${q.id})">编辑</button>
          <button class="btn-sm btn-danger" onclick="confirmDelQ(${q.id})">删除</button>
        </div></td>`;
      tbody.appendChild(tr);
      // 删除确认行
      if (q.confirming) {
        const cr = document.createElement('tr');
        cr.className = 'row-confirm no-hover';
        cr.innerHTML = `<td colspan="6"><div class="confirm-area">
          <span class="confirm-text">⚠️ 确认删除这道题目？</span>
          <button class="btn-sm btn-ghost" onclick="cancelDelQ(${q.id})">取消</button>
          <button class="btn-sm btn-danger" onclick="doDelQ(${q.id})">确认删除</button>
        </div></td>`;
        tbody.appendChild(cr);
      }
    }
  });
}
function toggleAddForm() { showingAddForm=!showingAddForm; renderQuestions(); }
function cancelAdd() { showingAddForm=false; renderQuestions(); }
function doAddQuestion() {
  const q=document.getElementById('na-question').value.trim();
  const a=document.getElementById('na-answer').value.trim();
  if(!q||!a){alert('题目和答案为必填项');return;}
  questions.unshift({
    id:Date.now(), subject:document.getElementById('na-subject').value||'未分类',
    topic:document.getElementById('na-topic').value||'未分类', question:q,
    options:document.getElementById('na-options').value, answer:a, editing:false, confirming:false
  });
  showingAddForm=false; renderQuestions();
}
function startEditQ(id) {
  questions.forEach(q=>{q.editing=false;q.confirming=false;});
  const q=questions.find(x=>x.id===id); if(q) q.editing=true; renderQuestions();
}
function cancelEditQ(id) { const q=questions.find(x=>x.id===id); if(q)q.editing=false; renderQuestions(); }
function doEditQ(id) {
  const q=questions.find(x=>x.id===id); if(!q)return;
  q.question=document.getElementById('eq-'+id).value;
  q.answer=document.getElementById('ea-'+id).value;
  q.options=document.getElementById('eo-'+id).value;
  q.editing=false; renderQuestions();
}
function confirmDelQ(id) {
  questions.forEach(q=>{q.confirming=false;q.editing=false;});
  const q=questions.find(x=>x.id===id); if(q)q.confirming=true; renderQuestions();
}
function cancelDelQ(id) { const q=questions.find(x=>x.id===id); if(q)q.confirming=false; renderQuestions(); }
function doDelQ(id) {
  const i=questions.findIndex(x=>x.id===id); if(i>-1)questions.splice(i,1); renderQuestions();
}

// ─── Init ─────────────────────────────────────────────────────────────────────
renderUsers(); renderQuestions();
